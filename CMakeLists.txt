cmake_minimum_required(VERSION 3.20)

project(credit_default_cpp_knn LANGUAGES CXX)

option(CPP_KNN_NATIVE "Enable native-CPU optimizations for local benchmarking" OFF)
option(CPP_KNN_SANITIZE "Enable available compiler sanitizers for tests" OFF)

# Bind every executable to the sources and build options used to produce it.
set(CPP_KNN_ID_FILES CMakeLists.txt src/cpp_knn/knn.hpp src/cpp_knn/knn.cpp
    src/cpp_knn/data_io.hpp src/cpp_knn/data_io.cpp src/cpp_knn/main.cpp)
set(CPP_KNN_ID_TEXT "")
foreach(source IN LISTS CPP_KNN_ID_FILES)
    file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/${source}" source_hash)
    string(APPEND CPP_KNN_ID_TEXT "${source}:${source_hash}\n")
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${source}")
endforeach()
string(SHA256 CPP_KNN_SOURCE_ID "${CPP_KNN_ID_TEXT}")

add_library(cpp_knn_core
    src/cpp_knn/data_io.cpp
    src/cpp_knn/knn.cpp
)
target_include_directories(cpp_knn_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/src)
target_compile_features(cpp_knn_core PUBLIC cxx_std_20)

add_executable(cpp_knn src/cpp_knn/main.cpp)
target_link_libraries(cpp_knn PRIVATE cpp_knn_core)
target_compile_features(cpp_knn PRIVATE cxx_std_20)
target_compile_definitions(cpp_knn PRIVATE CPP_KNN_SOURCE_ID="${CPP_KNN_SOURCE_ID}")

if(MSVC)
    target_compile_options(cpp_knn_core PRIVATE /W4 /permissive- /fp:precise)
    target_compile_options(cpp_knn PRIVATE /W4 /permissive- /fp:precise)
    if(CPP_KNN_NATIVE)
        include(CheckCXXSourceRuns)
        check_cxx_source_runs("#include <intrin.h>\nint main(){int r[4]; __cpuid(r,1); if (!(r[2]&(1<<27)) || !(r[2]&(1<<28))) return 1; if ((_xgetbv(0)&6)!=6) return 1; __cpuidex(r,7,0); return (r[1]&(1<<5))?0:1;}" CPP_KNN_HOST_AVX2)
        if(NOT CPP_KNN_HOST_AVX2)
            message(FATAL_ERROR "Native MSVC AVX2 requires a verified AVX2 CPU and OS support; use portable.")
        endif()
        target_compile_options(cpp_knn_core PRIVATE /O2 /arch:AVX2)
        target_compile_options(cpp_knn PRIVATE /O2 /arch:AVX2)
    endif()
else()
    target_compile_options(cpp_knn_core PRIVATE -Wall -Wextra -Wpedantic -ffp-contract=off)
    target_compile_options(cpp_knn PRIVATE -Wall -Wextra -Wpedantic -ffp-contract=off)
    if(CPP_KNN_NATIVE)
        target_compile_options(cpp_knn_core PRIVATE -O3 -march=native)
        target_compile_options(cpp_knn PRIVATE -O3 -march=native)
    endif()
endif()

if(CPP_KNN_NATIVE)
    include(CheckIPOSupported)
    check_ipo_supported(RESULT CPP_KNN_IPO_SUPPORTED OUTPUT CPP_KNN_IPO_REASON LANGUAGES CXX)
    if(CPP_KNN_IPO_SUPPORTED)
        set_property(TARGET cpp_knn_core cpp_knn PROPERTY INTERPROCEDURAL_OPTIMIZATION_RELEASE TRUE)
    else()
        message(STATUS "LTO unavailable: ${CPP_KNN_IPO_REASON}")
    endif()
    target_compile_definitions(cpp_knn_core PUBLIC CPP_KNN_NATIVE_BUILD=1)
endif()

if(CPP_KNN_SANITIZE)
    if(MSVC)
        target_compile_options(cpp_knn_core PUBLIC /fsanitize=address /Zi)
        target_link_options(cpp_knn_core INTERFACE /INCREMENTAL:NO)
    else()
        target_compile_options(cpp_knn_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
        target_link_options(cpp_knn_core INTERFACE -fsanitize=address,undefined)
    endif()
endif()

get_target_property(CPP_KNN_CORE_FLAGS cpp_knn_core COMPILE_OPTIONS)
string(REPLACE ";" " " CPP_KNN_CORE_FLAGS "${CPP_KNN_CORE_FLAGS}")
set(CPP_KNN_RECORDED_FLAGS "${CMAKE_CXX_FLAGS} ${CMAKE_CXX_FLAGS_RELEASE} ${CPP_KNN_CORE_FLAGS}")
target_compile_definitions(cpp_knn PRIVATE CPP_KNN_BUILD_FLAGS="${CPP_KNN_RECORDED_FLAGS}")
if(CPP_KNN_IPO_SUPPORTED)
    target_compile_definitions(cpp_knn PRIVATE CPP_KNN_LTO_ENABLED=1)
endif()

include(CTest)
if(BUILD_TESTING)
    add_executable(cpp_knn_tests tests/cpp_knn_tests.cpp)
    target_link_libraries(cpp_knn_tests PRIVATE cpp_knn_core)
    target_compile_features(cpp_knn_tests PRIVATE cxx_std_20)
    if(MSVC)
        target_compile_options(cpp_knn_tests PRIVATE /W4 /permissive- /fp:precise)
    else()
        target_compile_options(cpp_knn_tests PRIVATE -Wall -Wextra -Wpedantic -ffp-contract=off)
    endif()
    add_test(NAME cpp_knn_tests COMMAND cpp_knn_tests)
endif()

# MSVC's sanitizer is a toolchain DLL, not a system runtime. Keep the installed
# runtime beside local sanitized executables so ctest works outside a VS shell.
if(MSVC AND CPP_KNN_SANITIZE)
    get_filename_component(CPP_KNN_COMPILER_DIRECTORY "${CMAKE_CXX_COMPILER}" DIRECTORY)
    file(GLOB CPP_KNN_ASAN_RUNTIME "${CPP_KNN_COMPILER_DIRECTORY}/clang_rt.asan_dynamic-*.dll")
    if(NOT CPP_KNN_ASAN_RUNTIME)
        message(FATAL_ERROR "The selected MSVC toolchain has no AddressSanitizer runtime.")
    endif()
    foreach(target cpp_knn cpp_knn_tests)
        if(TARGET ${target})
            add_custom_command(TARGET ${target} POST_BUILD COMMAND ${CMAKE_COMMAND}
                -E copy_if_different ${CPP_KNN_ASAN_RUNTIME} $<TARGET_FILE_DIR:${target}>)
        endif()
    endforeach()
endif()
